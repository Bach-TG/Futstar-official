# ============================================================
#  CHROME DRIVER
# ============================================================
def make_driver() -> webdriver.Chrome:
    options = Options()
    options.add_argument("--headless=new")
    options.add_argument("--no-sandbox")
    options.add_argument("--disable-dev-shm-usage")
    options.add_argument("--disable-gpu")
    options.add_argument("--window-size=1920,1080")
    options.add_argument("--hide-scrollbars")

    chrome_bin = os.getenv("CHROME_BIN", "/usr/bin/chromium")
    options.binary_location = chrome_bin
    service = Service(os.getenv("CHROMEDRIVER", "/usr/bin/chromedriver"))

    return webdriver.Chrome(service=service, options=options)


# ============================================================
# HELPERS
# ============================================================
def _to_num(x: str) -> float | int:
    x = x.strip()
    if not x:
        return 0
    try:
        return int(x)
    except Exception:
        try:
            return float(x)
        except Exception:
            return 0


# ============================================================
#  TAB INDEX CONFIG
# ============================================================
TAB_INDEX: Dict[str, int] = {
    "General": 0,
    "Distribution": 1,
    "Attack": 2,
    "Defence": 3,
    "Discipline": 4,
}


# ============================================================
#  CLOCK READER
# ============================================================
def get_time_text(driver: webdriver.Chrome) -> str:
    # reading match clock directly
    selectors = [
        ".Opta-ClockContainer .Opta-Clock span",
        ".Opta-ClockContainer abbr",
    ]

    for sel in selectors:
        try:
            el = driver.find_element("css selector", sel)
            if el.text.strip():
                return el.text.strip()
        except:
            continue

    return ""


# ============================================================
#  EXECUTE_SCRIPT to get HTML of each tab
# ============================================================
def _extract_tab_html(driver, tab_name: str) -> str | None:
    idx = TAB_INDEX[tab_name] + 1  # nth-child = 1-based

    js = f"""
        const node = document.querySelector(
            "ul.Opta-TabbedContent > li:nth-child({idx}) table.Opta-Stats-Bars tbody"
        );
        return node ? node.innerHTML : null;
    """
    return driver.execute_script(js)


def parse_tab_html(html: str) -> dict:
    stats = {}
    if not html:
        return stats

    soup = BeautifulSoup(html, "html.parser")
    rows = soup.select("tr")

    for row in rows:
        cls = row.get("class", [])
        if not cls:
            continue

        cls = cls[0]
        if not cls.startswith("Opta-"):
            continue

        # Opta_shots_on_target → shots_on_target
        key = cls.split("_", 1)[1] if "_" in cls else cls.replace("Opta-", "")

        tds = row.select("td")
        if len(tds) < 2:
            continue

        left = _to_num(tds[0].get_text(strip=True))
        right = _to_num(tds[-1].get_text(strip=True))
        stats[key] = (left, right)

    return stats


def extract_tab(driver, tab_name: str) -> Dict[str, Any]:
    html = _extract_tab_html(driver, tab_name)
    stats = parse_tab_html(html)
    print(f"[SNAPSHOT_JS] {tab_name}: {stats}")
    return stats


def normalize_time(
    raw: str,
    is_halftime: bool,
    in_second_half: bool,
) -> str:
    """
    - Trước HT (is_halftime=False, in_second_half=False):
        >45:00 → 45:00 + mm:ss
    - Đang HT: raw sẽ là "HT" (xử lý trước khi gọi hàm này).
    - Sau HT (in_second_half=True):
        >90:00 → 90:00 + mm:ss
        còn lại → giữ nguyên mm:ss
    """
    raw = raw.strip()

    # "HT" / "FT" không xử lý ở đây
    if raw == "HT":
        return "Half Time"
    if raw == "FT":
        return "Full Time"

    if ":" not in raw:
        return raw

    try:
        m_str, s_str = raw.split(":")
        m = int(m_str)
        s = int(s_str)
    except Exception:
        return raw

    # FIRST HALF (trước HT)
    if not in_second_half:
        # nếu > 45:00 → coi như bù giờ hiệp 1
        total = m * 60 + s
        base = 45 * 60
        if total > base:
            extra = total - base
            ex_m = extra // 60
            ex_s = extra % 60
            return f"45:00 + {ex_m}:{ex_s:02d}"
        else:
            return f"{m}:{s:02d}"

    # SECOND HALF (sau HT)
    total = m * 60 + s
    base90 = 90 * 60
    if total > base90:
        extra = total - base90
        ex_m = extra // 60
        ex_s = extra % 60
        return f"90:00 + {ex_m}:{ex_s:02d}"

    return f"{m}:{s:02d}"


# ============================================================
#  MAIN CRAWLER STREAM
# ============================================================
async def crawl_match_data_stream(
    match_url: str,
    db_name: str,
) -> AsyncGenerator[tuple[str, bool, TeamMomentumStats, TeamMomentumStats], None]:
    await mongo_client.initialize(db_name=db_name)

    driver = make_driver()
    driver.get(match_url)

    tabs = ["General", "Distribution", "Attack", "Defence", "Discipline"]

    print("[Crawler] ⚡ Start loop")

    is_halftime = False  # đang ở HT hay không
    in_second_half = False  # đã sang hiệp 2 chưa
    last_emitted_time = None  # tránh emit trùng time_in_match

    try:
        while True:
            try:
                # -----------------------------------
                # CLOCK
                # -----------------------------------
                raw_time = get_time_text(driver)
                if not raw_time:
                    await asyncio.sleep(0.1)
                    continue

                raw_time = raw_time.strip()
                print(f"\n[RAW CLOCK] {raw_time}")

                # ========== FULL TIME ==========
                if raw_time == "FT":
                    print("[Crawler] 🔴 Full-time detected → stopping crawler.")
                    # emit 1 tick FT (optional: stats rỗng)
                    home_empty = TeamMomentumStats(
                        time_in_match="Full Time", stats=MomentumUsedStats()
                    )
                    away_empty = TeamMomentumStats(
                        time_in_match="Full Time", stats=MomentumUsedStats()
                    )
                    yield ("Full Time", False, home_empty, away_empty)
                    break

                # ========== HALF TIME ==========
                if raw_time == "HT":
                    if not is_halftime:
                        is_halftime = True
                        print("[Crawler] 🟡 Halftime detected.")
                        home_empty = TeamMomentumStats(
                            time_in_match="Half Time", stats=MomentumUsedStats()
                        )
                        # is_halftime = True ở tick này

                        yield ("Half Time", True, home_empty, away_empty)
                    # Trong thời gian HT: không crawl stats, chỉ đợi đồng hồ đổi.
                    await asyncio.sleep(1.0)
                    continue

                # Nếu đang HT mà đồng hồ trở lại dạng MM:SS → bắt đầu hiệp 2
                if is_halftime and raw_time not in {"HT", "FT"}:
                    is_halftime = False
                    in_second_half = True
                    print(f"[Crawler] 🟢 Second half started at raw={raw_time}")

                # -----------------------------------
                # Chuẩn hoá time theo luật bù giờ
                # -----------------------------------
                norm_time = normalize_time(
                    raw=raw_time,
                    is_halftime=is_halftime,
                    in_second_half=in_second_half,
                )

                print(f"[CLOCK] raw={raw_time}  → normalized={norm_time}")

                # # Nếu trùng time_in_match với tick trước đó → bỏ qua để tránh trùng
                # if norm_time == last_emitted_time:
                #     await asyncio.sleep(0.3)
                #     continue

                # -----------------------------------
                # Extract all tabs
                # -----------------------------------
                stats_all: Dict[str, Dict[str, Any]] = {}
                for tab in tabs:
                    stats_all[tab] = extract_tab(driver, tab)

                attack = stats_all["Attack"]
                defence = stats_all["Defence"]
                discipline = stats_all["Discipline"]
                distribution = stats_all["Distribution"]
                general = stats_all["General"]

                # -----------------------------------
                # Build momentum objects
                # -----------------------------------
                def build(idx: int) -> MomentumUsedStats:
                    return MomentumUsedStats(
                        goals=attack.get("goals", (0, 0))[idx],
                        shots_on_target=attack.get("shots_on_target", (0, 0))[idx],
                        shots_inside_box=attack.get("shots_inside_box", (0, 0))[idx],
                        shots_headed=attack.get("shots_headed", (0, 0))[idx],
                        shots_outside_box=attack.get("shots_outside_box", (0, 0))[idx],
                        shots_blocked=attack.get("shots_blocked", (0, 0))[idx],
                        key_passes=attack.get("key_passes", (0, 0))[idx],
                        through_balls=distribution.get("through_balls", (0, 0))[idx],
                        penalty_area_entries=distribution.get(
                            "penalty_area_entries", (0, 0)
                        )[idx],
                        open_play_crosses=distribution.get("open_play_crosses", (0, 0))[
                            idx
                        ],
                        final_thirds_entries=distribution.get(
                            "final_thirds_entries", (0, 0)
                        )[idx],
                        dribbles_success=general.get("dribbles_success", (0, 0))[idx],
                        corners_won=general.get("corners_won", (0, 0))[idx],
                        possession_lost_defensive=general.get(
                            "possession_lost_defensive", (0, 0)
                        )[idx],
                        possession_lost_midfield=general.get(
                            "possession_lost_midfield", (0, 0)
                        )[idx],
                        offsides=general.get("offsides", (0, 0))[idx],
                        ball_recoveries_attacking=defence.get(
                            "ball_recoveries_attacking", (0, 0)
                        )[idx],
                        interceptions=defence.get("interceptions", (0, 0))[idx],
                        tackles=defence.get("tackles", (0, 0))[idx],
                        ball_recoveries_midfield=defence.get(
                            "ball_recoveries_midfield", (0, 0)
                        )[idx],
                        ball_recoveries_defensive=defence.get(
                            "ball_recoveries_defensive", (0, 0)
                        )[idx],
                        clearances=defence.get("clearances", (0, 0))[idx],
                        fouls_conceded=discipline.get("fouls_conceded", (0, 0))[idx],
                        fouls_won=general.get("fouls_won", (0, 0))[idx],
                        cards_yellow=discipline.get("cards_yellow", (0, 0))[idx],
                        cards_red=discipline.get("cards_red", (0, 0))[idx],
                    )

                home = build(0)
                away = build(1)

                # -----------------------------------
                # YIELD TICK
                # -----------------------------------
                # last_emitted_time = norm_time

                yield (
                    norm_time,
                    False,  # is_halftime chỉ true đúng tick HT ở trên
                    TeamMomentumStats(time_in_match=norm_time, stats=home),
                    TeamMomentumStats(time_in_match=norm_time, stats=away),
                )

                await asyncio.sleep(0.85)

            except Exception as e:
                print(f"[Crawler] ⚠ Soft error: {e}")
                await asyncio.sleep(0.5)
    finally:
        driver.quit()
        print("[Crawler] Chrome closed.")
